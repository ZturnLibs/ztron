`vite-plugin.ts` —— 导出 `isLocalHostname`，重写中间件：

```ts
/** True for hostnames the dev server may serve (localhost variants only). */
export function isLocalHostname(rawHost: string): boolean {
  let host = rawHost.trim().toLowerCase();
  if (host.startsWith("[")) {
    // [::1]:port → ::1
    const end = host.indexOf("]");
    host = end === -1 ? host.slice(1) : host.slice(1, end);
  } else {
    const colon = host.lastIndexOf(":");
    if (colon !== -1 && host.indexOf(":") === colon) host = host.slice(0, colon);
  }
  return host === "localhost" || host === "127.0.0.1" || host === "::1" ||
    host.endsWith(".localhost");
}

// configureServer 内（替换现有 CORS 中间件体）：
server.middlewares.use(
  (
    req: { headers?: Record<string, string | string[] | undefined> },
    res: {
      statusCode?: number;
      setHeader: (k: string, v: string) => void;
      end: (chunk?: string) => void;
    },
    next: () => void,
  ) => {
    const hostHeader = req.headers?.host;
    const rawHost = Array.isArray(hostHeader) ? hostHeader[0] : hostHeader;
    // DNS-rebinding guard (tauri cc9d522c6): only localhost variants may hit us.
    if (!rawHost || !isLocalHostname(rawHost)) {
      res.statusCode = 403;
      res.end("Forbidden: non-local Host header");
      return;
    }
    const originHeader = req.headers?.origin;
    const origin = Array.isArray(originHeader) ? originHeader[0] : originHeader;
    // Same-origin requests (WKWebView module loads) carry no Origin header —
    // keep the wildcard so ESM loads work. Foreign origins get no ACAO,
    // so the response is unreadable cross-site.
    if (origin && !isLocalHostname(origin.replace(/^https?:\/\//, ""))) {
      next();
      return;
    }
    res.setHeader(
      "Access-Control-Allow-Origin",
      origin && origin.startsWith("http") ? origin : "*",
    );
    next();
  },
);
```

```c
/* Liquid Glass (macOS 26+): NSGlassEffectView; returns false below 26. */
static bool window_apply_liquid_glass(void *wnd, const char *name,
                                       const char *color_hex, double radius,
                                       int interactive) {
  Class cls = objc_getClass("NSGlassEffectView");
  if (!cls) return false; /* macOS < 26: caller falls back to vibrancy */
  id content = OBJC_MSG(id(*)(id, SEL), (id)wnd, sel_registerName("contentView"));
  if (!content) return false;
  id v = OBJC_MSG(id(*)(id, SEL), (id)cls, sel_registerName("new"));
  ZtRect b = ((ZtRect(*)(id, SEL))objc_msgSend)(content, sel_registerName("bounds"));
  ((void(*)(id, SEL, ZtRect))objc_msgSend)(v, sel_registerName("setFrame:"), b);
  /* NSGlassEffectViewStyle: Regular=0, Clear=1 (SDK header). */
  OBJC_MSG(void(*)(id, SEL, long), v, sel_registerName("setStyle:"),
           (long)(strcmp(name, "liquidGlassClear") == 0 ? 1 : 0));
  if (interactive && ((BOOL(*)(id, SEL, SEL))objc_msgSend)(
        v, sel_registerName("respondsToSelector:"),
        sel_registerName("setInteractive:"))) {
    /* macOS 27+; the 26 SDK has no such property. */
    OBJC_MSG(void(*)(id, SEL, BOOL), v, sel_registerName("setInteractive:"),
             (BOOL)YES);
  }
  if (color_hex && color_hex[0]) {
    id cs = (id)objc_getClass("NSColor");
    id color = OBJC_MSG(id(*)(id, SEL, const char*), cs,
                        sel_registerName("colorWithSRGBRed:green:blue:alpha:"), 0, 0, 0, 0);
    /* hex parse: #RRGGBB → 0..1 components */
    unsigned r = 0, g = 0, b2 = 0;
    if (sscanf(color_hex, "#%02x%02x%02x", &r, &g, &b2) == 3) {
      color = ((id(*)(id, SEL, double, double, double, double))objc_msgSend)(
          cs, sel_registerName("colorWithSRGBRed:green:blue:alpha:"),
          r / 255.0, g / 255.0, b2 / 255.0, 1.0);
    }
    OBJC_MSG(void(*)(id, SEL, id), v, sel_registerName("setTintColor:"), color);
  }
  if (radius > 0) {
    ((void(*)(id, SEL, double))objc_msgSend)(v, sel_registerName("setCornerRadius:"), radius);
  }
  ((void(*)(id, SEL, id, unsigned long, id))objc_msgSend)(
      content, sel_registerName("addSubview:positioned:relativeTo:"), v,
      2UL /* NSWindowBelow */, (id)0);
  return true;
}
```

```ts
/** Normalizes a {@linkcode JsImage} into a registered {@linkcode Image}. */
async function normalizeIcon(icon: JsImage): Promise<Image> {
  if (icon instanceof Image) return icon;
  if (typeof icon === "string") return Image.fromPath(icon);
  const bytes =
    icon instanceof Uint8Array
      ? icon
      : new Uint8Array(icon instanceof ArrayBuffer ? icon : icon);
  return Image.fromBytes(bytes);
}

async setIcon(icon: JsImage | null): Promise<void> {
  const img = icon === null ? null : await normalizeIcon(icon);
  await invoke("plugin:window|set_icon", {
    label: this.label,
    image_id: img ? img.rid : -1,
  });
}
```
